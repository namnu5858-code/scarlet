async function capturePhoto() { if (!cameraVideo.videoWidth || !cameraVideo.videoHeight) return; const canvas = document.createElement("canvas"); const size = Math.min(cameraVideo.videoWidth, cameraVideo.videoHeight); canvas.width = size; canvas.height = size; const ctx = canvas.getContext("2d"); const sx = (cameraVideo.videoWidth - size) / 2; const sy = (cameraVideo.videoHeight - size) / 2; ctx.drawImage(cameraVideo, sx, sy, size, size, 0, 0, size, size); const dataUrl = canvas.toDataURL("image/jpeg", 0.9); enterPreviewMode(dataUrl); }
