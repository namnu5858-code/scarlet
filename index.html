shutterBtn.addEventListener("click", async () => { if (!videoFeed.videoWidth || !videoFeed.videoHeight) return; const canvas = document.createElement("canvas"); const size = Math.min(videoFeed.videoWidth, videoFeed.videoHeight); canvas.width = size; canvas.height = size; const ctx = canvas.getContext("2d"); const sx = (videoFeed.videoWidth - size) / 2; const sy = (videoFeed.videoHeight - size) / 2; ctx.drawImage(videoFeed, sx, sy, size, size, 0, 0, size, size); const dataUrl = canvas.toDataURL("image/jpeg", 0.9); openPreviewWithImage(dataUrl); });
